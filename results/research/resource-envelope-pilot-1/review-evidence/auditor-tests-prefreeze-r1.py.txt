"""Negative-capable synthetic tests for the independent resource input auditor.

All fixtures here are hand-written n=2 examples with synthetic IDs. They are
not members of the selected 16..512 scientific cohort and are never stored in
its corpus directories.
"""
from __future__ import annotations

import copy
import hashlib
import json
import unittest

from lib import resource_envelope_audit as audit


SOURCE_PI_2 = (b"-- Generated by resource-envelope-pilot-1; exact scientific input.\n"
               b"namespace ResourceEnvelopePilot1\n\n"
               b"def pi002 : Sort 1 := (forall (x000 : Sort 0), (forall (x001 : Sort 0), Sort 0))\n\n"
               b"end ResourceEnvelopePilot1\n")
SOURCE_LET_2 = (b"-- Generated by resource-envelope-pilot-1; exact scientific input.\n"
                b"namespace ResourceEnvelopePilot1\n\n"
                b"def let002 : Sort 1 := (let x000 : Sort 1 := Sort 0; (let x001 : Sort 1 := x000; x001))\n\n"
                b"end ResourceEnvelopePilot1\n")
SOURCE_LET_3 = (b"-- Generated by resource-envelope-pilot-1; exact scientific input.\n"
                b"namespace ResourceEnvelopePilot1\n\n"
                b"def let003 : Sort 1 := (let x000 : Sort 1 := Sort 0; (let x001 : Sort 1 := x000; (let x002 : Sort 1 := x001; x002)))\n\n"
                b"end ResourceEnvelopePilot1\n")


META = {
    "meta": {
        "exporter": {"name": "lean4export", "version": "3.1.0"},
        "format": {"version": "3.1.0"},
        "lean": {"githash": "f72c35b3f637c8c6571d353742168ab66cc22c00",
                 "version": "4.29.1"},
    }
}


def _rows(family: str) -> list[dict]:
    name = f"{family}002"
    rows = [META,
            {"in": 1, "str": {"pre": 0, "str": "ResourceEnvelopePilot1"}},
            {"in": 2, "str": {"pre": 1, "str": name}},
            {"in": 3, "str": {"pre": 0, "str": "x000"}},
            {"in": 4, "str": {"pre": 0, "str": "x001"}},
            {"il": 1, "succ": 0},
            {"ie": 0, "sort": 0}]
    if family == "pi":
        rows.extend([
            {"ie": 1, "forallE": {"binderInfo": "default", "body": 0,
                                    "name": 4, "type": 0}},
            {"ie": 2, "forallE": {"binderInfo": "default", "body": 1,
                                    "name": 3, "type": 0}},
            {"ie": 3, "sort": 1},
        ])
        value_id, type_id = 2, 3
    else:
        rows.extend([
            {"ie": 1, "sort": 1},
            {"ie": 2, "bvar": 0},
            {"ie": 3, "letE": {"body": 2, "name": 4, "nondep": False,
                                 "type": 1, "value": 2}},
            {"ie": 4, "letE": {"body": 3, "name": 3, "nondep": False,
                                 "type": 1, "value": 0}},
            {"ie": 5, "sort": 1},
        ])
        value_id, type_id = 4, 5
    rows.append({"def": {"all": [2], "hints": {"regular": 1},
                          "levelParams": [], "name": 2, "safety": "safe",
                          "type": type_id, "value": value_id}})
    return rows


def _export_bytes(family: str, mutate=None) -> bytes:
    rows = copy.deepcopy(_rows(family))
    if mutate:
        mutate(rows)
    return b"".join(json.dumps(row, separators=(",", ":"), ensure_ascii=False).encode() + b"\n"
                   for row in rows)


def _case(family: str, export: bytes | None = None, source: bytes | None = None) -> tuple[dict, bytes, bytes]:
    ident = f"synthetic-{family}-two"
    source = source if source is not None else (SOURCE_PI_2 if family == "pi" else SOURCE_LET_2)
    export = export if export is not None else _export_bytes(family)
    dag = 3 if family == "pi" else 5
    expanded = 5 if family == "pi" else 7
    row = {
        "schema_version": 1,
        "id": ident,
        "ordinal": 0,
        "family": family,
        "size": 2,
        "declaration": f"ResourceEnvelopePilot1.{family}002",
        "expected_type": "Sort 1",
        "source": {"path": f"corpus/sources/{ident}.lean", "bytes": len(source),
                   "sha256": hashlib.sha256(source).hexdigest()},
        "export": {"path": f"corpus/exports/{ident}.ndjson", "bytes": len(export),
                   "sha256": hashlib.sha256(export).hexdigest()},
        "metrics": {"serialized_bytes": len(export), "reachable_dag_nodes": dag,
                    "expanded_value_nodes": expanded, "binder_depth": 2,
                    "dependency_declarations": 0},
    }
    return row, export, source

def _replace_record(rows, kind, fn):
    for row in rows:
        if kind in row:
            fn(row[kind])
            return
    raise AssertionError(kind)


class ResourceEnvelopeAuditTests(unittest.TestCase):
    def assertAuditCode(self, code, row, export, source):
        with self.assertRaises(audit.AuditError) as caught:
            audit.audit_case(row, export, source)
        self.assertEqual(caught.exception.code, code)

    def test_source_renderer_matches_handwritten_disjoint_templates(self):
        self.assertEqual(audit.render_source("pi", 2), SOURCE_PI_2)
        self.assertEqual(audit.render_source("let", 2), SOURCE_LET_2)
        self.assertEqual(audit.render_source("let", 3), SOURCE_LET_3)

    def test_synthetic_pi_and_let_formulas_pass(self):
        for family in ("pi", "let"):
            with self.subTest(family=family):
                row, export, source = _case(family)
                result = audit.audit_case(row, export, source)
                self.assertEqual(result["status"], "PASS")
                self.assertEqual(result["expanded_value_nodes"], 5 if family == "pi" else 7)
                self.assertEqual(result["reachable_dag_nodes"], 3 if family == "pi" else 5)
                self.assertEqual(result["binder_depth"], 2)

    def test_consistent_valid_type_and_metrics_do_not_hide_wrong_family_value(self):
        row, export, source = _let_three_wrong_outer_value()
        self.assertAuditCode("E_EXPECTATION", row, export, source)

    def test_wrong_type_is_rejected(self):
        row, export, source = _case("pi", _export_bytes("pi", lambda rows:
            _replace_record(rows, "def", lambda d: d.update(type=0))))
        row["export"]["bytes"] = len(export)
        row["export"]["sha256"] = hashlib.sha256(export).hexdigest()
        row["metrics"]["serialized_bytes"] = len(export)
        self.assertAuditCode("E_EXPECTED_TYPE", row, export, source)

    def test_wrong_value_even_with_updated_hash_and_size_is_rejected(self):
        def mutate(rows):
            for row in rows:
                if row.get("ie") == 1:
                    row["forallE"]["binderInfo"] = "instImplicit"
                    return
        row, export, source = _case("pi", _export_bytes("pi", mutate))
        row["export"].update(bytes=len(export), sha256=hashlib.sha256(export).hexdigest())
        row["metrics"]["serialized_bytes"] = len(export)
        self.assertAuditCode("E_BINDER_INFO", row, export, source)

    def test_unsafe_definition_is_rejected(self):
        row, export, source = _case("pi", _export_bytes("pi", lambda rows:
            _replace_record(rows, "def", lambda d: d.update(safety="unsafe"))))
        row["export"].update(bytes=len(export), sha256=hashlib.sha256(export).hexdigest())
        row["metrics"]["serialized_bytes"] = len(export)
        self.assertAuditCode("E_UNSAFE", row, export, source)

    def test_axiom_or_opaque_extra_declaration_is_rejected(self):
        row, export, source = _case("pi")
        rows = _rows("pi")
        rows.insert(-1, {"axiom": {"name": 2, "type": 3}})
        export = b"".join(json.dumps(x, separators=(",", ":")).encode() + b"\n" for x in rows)
        row["export"].update(bytes=len(export), sha256=hashlib.sha256(export).hexdigest())
        row["metrics"]["serialized_bytes"] = len(export)
        self.assertAuditCode("E_EXPORT_RECORD", row, export, source)

    def test_partial_or_unknown_definition_field_is_rejected(self):
        row, export, source = _case("pi", _export_bytes("pi", lambda rows:
            _replace_record(rows, "def", lambda d: d.update(partial=True))))
        row["export"].update(bytes=len(export), sha256=hashlib.sha256(export).hexdigest())
        row["metrics"]["serialized_bytes"] = len(export)
        self.assertAuditCode("E_SCHEMA", row, export, source)

    def test_let_nondep_must_be_false(self):
        row, export, source = _case("let", _export_bytes("let", lambda rows:
            _replace_record(rows, "letE", lambda d: d.update(nondep=True))))
        row["export"].update(bytes=len(export), sha256=hashlib.sha256(export).hexdigest())
        row["metrics"]["serialized_bytes"] = len(export)
        self.assertAuditCode("E_LET_NONDEP", row, export, source)

    def test_no_universe_parameters(self):
        row, export, source = _case("pi", _export_bytes("pi", lambda rows:
            _replace_record(rows, "def", lambda d: d.update(levelParams=["u"]))))
        row["export"].update(bytes=len(export), sha256=hashlib.sha256(export).hexdigest())
        row["metrics"]["serialized_bytes"] = len(export)
        self.assertAuditCode("E_LEVEL_PARAMS", row, export, source)

    def test_definition_hints_are_exact(self):
        row, export, source = _case("pi", _export_bytes("pi", lambda rows:
            _replace_record(rows, "def", lambda d: d.update(hints={"abbrev": True}))))
        row["export"].update(bytes=len(export), sha256=hashlib.sha256(export).hexdigest())
        row["metrics"]["serialized_bytes"] = len(export)
        self.assertAuditCode("E_SCHEMA", row, export, source)

    def test_source_mutation_fails_even_with_rebound_hash(self):
        row, export, source = _case("pi")
        changed = source.replace(b"Sort 1", b"Sort 0", 1)
        row["source"].update(bytes=len(changed), sha256=hashlib.sha256(changed).hexdigest())
        self.assertAuditCode("E_SOURCE", row, export, changed)

    def test_byte_hash_and_boolean_integer_mutations_fail(self):
        row, export, source = _case("pi")
        row["metrics"]["binder_depth"] = True
        self.assertAuditCode("E_NATURAL", row, export, source)
        row, export, source = _case("pi")
        row["export"]["sha256"] = "0" * 64
        self.assertAuditCode("E_HASH", row, export, source)

    def test_duplicate_and_dangling_expression_records_fail(self):
        for mutate, expected in (
            (lambda rows: rows.insert(-1, next(r for r in rows if r.get("ie") == 0)), "E_EXPR"),
            (lambda rows: next(r for r in rows if r.get("ie") == 1).__setitem__("ie", 99), "E_EXPR_REF"),
        ):
            row, export, source = _case("pi", _export_bytes("pi", mutate))
            row["export"].update(bytes=len(export), sha256=hashlib.sha256(export).hexdigest())
            row["metrics"]["serialized_bytes"] = len(export)
            self.assertAuditCode(expected, row, export, source)

    def test_loose_bound_variable_is_rejected_before_formula_comparison(self):
        row, export, source = _case("let", _export_bytes("let", lambda rows:
            _replace_record(rows, "bvar", lambda d: None)))
        # Rewrite the sole bvar to index 2: it is outside either n=2 let scope.
        records = [json.loads(line) for line in export.splitlines()]
        next(r for r in records if "bvar" in r)["bvar"] = 2
        export = b"".join(json.dumps(x, separators=(",", ":")).encode() + b"\n" for x in records)
        row["export"].update(bytes=len(export), sha256=hashlib.sha256(export).hexdigest())
        row["metrics"]["serialized_bytes"] = len(export)
        self.assertAuditCode("E_SCOPE", row, export, source)

    def test_extra_dead_expression_fails(self):
        row, export, source = _case("pi", _export_bytes("pi", lambda rows:
            rows.insert(-1, {"ie": 10, "sort": 0})))
        row["export"].update(bytes=len(export), sha256=hashlib.sha256(export).hexdigest())
        row["metrics"]["serialized_bytes"] = len(export)
        self.assertAuditCode("E_DEAD_RECORD", row, export, source)

    def test_duplicate_name_prefix_cycle_and_bad_level_fail(self):
        row, export, source = _case("pi", _export_bytes("pi", lambda rows:
            rows[1].update(str={"pre": 2, "str": "Cycle"})))
        row["export"].update(bytes=len(export), sha256=hashlib.sha256(export).hexdigest())
        row["metrics"]["serialized_bytes"] = len(export)
        self.assertAuditCode("E_CYCLE", row, export, source)
        row, export, source = _case("pi", _export_bytes("pi", lambda rows:
            rows[5].update(succ=True)))
        row["export"].update(bytes=len(export), sha256=hashlib.sha256(export).hexdigest())
        row["metrics"]["serialized_bytes"] = len(export)
        self.assertAuditCode("E_NATURAL", row, export, source)

    def test_exact_selected_grid_is_pi_then_let_at_six_frozen_sizes(self):
        expected = audit._expected_cases()
        sizes = (16, 32, 64, 128, 256, 512)
        self.assertEqual([(x["id"], x["ordinal"], x["family"], x["size"]) for x in expected],
                         [(f"rep1-{family}-{size:03d}", ordinal, family, size)
                          for ordinal, (family, size) in enumerate(
                              [(family, size) for family in ("pi", "let") for size in sizes])])

    def test_constant_or_hidden_expression_tag_is_rejected(self):
        row, export, source = _case("pi", _export_bytes("pi", lambda rows:
            rows.insert(-1, {"ie": 10, "const": {"name": 2, "levels": []}})))
        row["export"].update(bytes=len(export), sha256=hashlib.sha256(export).hexdigest())
        row["metrics"]["serialized_bytes"] = len(export)
        self.assertAuditCode("E_EXPR_TAG", row, export, source)

    def test_index_and_corpus_membership_helpers_reject_order_and_extras(self):
        expected = [
            {"id": "synthetic-a", "ordinal": 0, "family": "pi", "size": 2},
            {"id": "synthetic-b", "ordinal": 1, "family": "let", "size": 3},
        ]
        refs = [{**row, "case": {"path": f"corpus/cases/{row['id']}.json",
                                  "bytes": 1, "sha256": "0" * 64}} for row in expected]
        self.assertEqual(len(audit._check_index_rows(refs, expected)), 2)
        with self.assertRaises(audit.AuditError) as caught:
            audit._check_index_rows(list(reversed(refs)), expected)
        self.assertEqual(caught.exception.code, "E_INDEX_ORDER")
        duplicate = copy.deepcopy(refs)
        duplicate[1]["id"] = duplicate[0]["id"]
        with self.assertRaises(audit.AuditError):
            audit._check_index_rows(duplicate, expected)
        with self.assertRaises(audit.AuditError) as caught:
            audit._check_index_rows(refs[:1], expected)
        self.assertEqual(caught.exception.code, "E_INDEX")
        bool_ordinal = copy.deepcopy(refs)
        bool_ordinal[1]["ordinal"] = True
        with self.assertRaises(audit.AuditError):
            audit._check_index_rows(bool_ordinal, expected)
        with self.assertRaises(audit.AuditError) as caught:
            audit._check_members({"synthetic-a", "extra"}, {"synthetic-a", "synthetic-b"}, "fixture")
        self.assertEqual(caught.exception.code, "E_CORPUS_MEMBERSHIP")


if __name__ == "__main__":
